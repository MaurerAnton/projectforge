package org.projectforge.framework.persistence.utils

import de.micromata.merlin.excel.importer.ImportedElement
import de.micromata.merlin.excel.importer.ImportedSheet
import org.projectforge.framework.DisplayNameCapable

/**
 * Clase de elemento importado personalizada que formatea valores como cadenas, particularmente para nombres mostrados.
 * Esta clase extiende la clase genérica ImportedElement para proporcionar funcionalidad específica
 * para manejar datos importados con nombres mostrados.
 * @param T El tipo de los datos importados.
 * @param importedSheet La hoja desde la cual se importan los datos.
 * @param row El número de fila en la hoja.
 * @param clazz El tipo de clase de los datos importados.
 * @param diffProperties Propiedades opcionales para rastrear diferencias en los datos importados.
 *                       Lista de nombres de propiedades que se usarán para mostrar cambios de propiedades.
 */
open class MyImportedElement<T>(
    importedSheet: ImportedSheet<T>,
    row: Int,
    clazz: Class<T>,
    vararg diffProperties: String
) : ImportedElement<T>(importedSheet, row, clazz, *diffProperties) {

    override fun valueAsString(value: Any?): String? {
        if (value == null)
            return null
        return when (value) {
            is DisplayNameCapable -> value.displayName
            else -> value.toString()
        }
    }
}
