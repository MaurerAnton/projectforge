    /**
     * Normaliza un número de teléfono a un formato estandarizado y legible.
     *
     * Formato: +[País] [Área] [Número][-Extensión]
     *
     * Ejemplos:
     * - "0561 316793-0" → "+49 561 316793-0"
     * - "+49 (0) 561 / 316793-0" → "+49 561 316793-0"
     * - "0170 1234567" → "+49 170 1234567"
     * - "+1 415 555-1234" → "+1 415 5551234"
     *
     * @param phoneNumber El número de teléfono a normalizar (puede contener espacios, barras, paréntesis, etc.)
     * @param defaultCountryPrefix El prefijo de país predeterminado a usar si no hay ninguno presente (por defecto: desde configuración o "+49")
     * @return Número de teléfono normalizado en formato "+CC AAA NNNNNNN[-EXT]" o null si la entrada es nula/vacía
     */
    fun normalizePhoneNumber(
        phoneNumber: String?,
        defaultCountryPrefix: String = getDefaultCountryPrefix()
    ): String? {
        if (phoneNumber.isNullOrBlank()) {
            return null
        }

        var cleaned = phoneNumber.trim()

        // Eliminar prefijos como "Tel:", "Phone:", "Fax:", etc.
        cleaned = cleaned.replace(Regex("^(?:Tel\\.?:|Telefon:|Phone:|Mobil:|Mobile:|Fax:)\\s*", RegexOption.IGNORE_CASE), "")

        // Eliminar caracteres de control UTF
        cleaned = cleaned.replace("\\p{C}".toRegex(), "")

        // Eliminar (0) en +49 (0) 123456789
        cleaned = cleaned.replace(Regex("\\(0\\)"), "")

        // Eliminar paréntesis y barras al inicio para la detección del código de país
        cleaned = cleaned.replace(Regex("[/()]"), "")

        if (cleaned.isEmpty()) {
            return null
