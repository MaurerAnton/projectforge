                                  created                  TIMESTAMP WITHOUT TIME ZONE,
                                  last_update              TIMESTAMP WITHOUT TIME ZONE,
                                  owner_fk                 INTEGER,
                                  credential_id            CHARACTER VARYING(4000),
                                  display_name             CHARACTER VARYING(1000),
                                  attested_credential_data CHARACTER VARYING(10000),

    );
