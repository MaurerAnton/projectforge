import PropTypes from 'prop-types';
import React from 'react';

// Validar automáticamente los campos de entrada anidados en el interior. (Solo primer nivel)
function ValidationManager({ children, customValidation }) {
    const extraLabel = null;

    return (
        <>
            {React.Children.map(children, (child) => {
                if (!child) {
                    return child;
                }
                const { props: childProps } = child;
                const {
                    checked,
                    required,
                    maxLength,
                    value,
                } = childProps;

                let valid = true;
                let { additionalLabel } = childProps;

                // Validar required y maxLength
                if (
                    (required && !(value || checked))
                    || (maxLength && value && value.length > maxLength)
                ) {
                    valid = false;
                }

                // Verificar una validación personalizada.
                if (customValidation) {
                    valid = false;

                    if (additionalLabel) {
                        // Agregar el mensaje de validación a la etiqueta adicional si existe
                        additionalLabel += ` - ${customValidation.message}`;
                    } else {
                        // o si no, establecer la etiqueta adicional al mensaje de validación.
                        additionalLabel = customValidation.message;
                    }
                }

                return {
                    ...child,
                    // Manipulando las props del hijo.
                    props: {
                        ...child.props,
                        additionalLabel,
                        color: valid ? childProps.color : 'danger',
                    },
                };
            })}
            {extraLabel}
        </>
    );
}

ValidationManager.propTypes = {
    children: PropTypes.node.isRequired,
    customValidation: PropTypes.shape({
        message: PropTypes.string,
    }),
};

export default ValidationManager;
