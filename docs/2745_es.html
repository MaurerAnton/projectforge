const categoryReducer = (state = initialCategoryState, { type, payload } = {}) => {
    switch (type) {
        case LIST_DISMISS_ERROR:
            return {
                ...state,
                error: undefined,
            };
        case LIST_INITIAL_CALL_BEGIN:
            return {
                ...state,
                isFetching: true,
                search: payload.search,
                error: undefined,
            };
        case LIST_FETCH_DATA_BEGIN:
            return {
                ...state,
                isFetching: true,
                error: undefined,
                variables: {
                    ...state.variables,
                    ...payload.variables,
                },
            };
        case LIST_CALL_SUCCESS:
            return {
                ...state,
                isFetching: false,
                lastQueriedFilter: JSON.stringify(payload.response.filter || state.filter),
                newlySwitched: false,
                ...payload.response,
            };
        case LIST_FETCH_FAILURE:
            return {
                ...state,
                isFetching: false,
                error: payload.error,
            };
        case LIST_FILTER_ADD: {
            const { filter } = state;

            return {
                ...state,
                filter: {
                    ...filter,
                    entries: [
                        ...filter.entries,
                        {
                            field: payload.fieldId,
                            isNew: true,
                        },
                    ],
                },
                newlySwitched: false,
            };
        }

            return {
                ...state,
                filter: {
                    ...filter,
