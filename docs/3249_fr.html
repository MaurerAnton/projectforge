package org.projectforge.web.wicket.bootstrap;

/**
 * Utilisé pour définir la valeur de l'attribut de classe pour les éléments (tailles de grille bootstrap).
 * <ul>
 * <li>COL25 est un alias pour SPAN3</li>
 * <li>COL33 est un alias pour SPAN4</li>
 * <li>COL50 est un alias pour SPAN6</li>
 * <li>COL66 est un alias pour SPAN8</li>
 * <li>COL75 est un alias pour SPAN9</li>
 * <li>COL100 est un alias pour SPAN12</li>
 * </ul>
 *
 * @author Kai Reinhard (k.reinhard@micromata.de)
 */
public enum GridSize
{
  SPAN1(1), SPAN2(2), SPAN3(3), SPAN4(4), SPAN6(6), SPAN8(8), SPAN9(9), SPAN12(12), COL25(SPAN3), COL33(SPAN4), COL50(
      SPAN6), COL66(SPAN8), COL75(
          SPAN9), COL100(SPAN12);

  private final String classAttrValue;

  private final int length;

  public String getClassAttrValue()
  {
    return classAttrValue;
  }

  /**
   * @return la longueur
   */
  public int getLength()
  {
    return length;
  }

  private GridSize(final int length)
  {
    this.length = length;
    // Pour maintenir des largeurs fluides sur les ordinateurs de bureau & tablettes, mais empiler verticalement sur les smartphones :
    // Remplacer .col-md-* par .col-sm-*
    //
    // Pour maintenir des largeurs fluides sur tous les appareils (sans empilement) :
    // Remplacer .col-md-* par .col-xs-*
    this.classAttrValue = "col-sm-" + length;
  }

  private GridSize(final GridSize master)
  {
    this.length = master.length;
    this.classAttrValue = master.classAttrValue;
  }

  public static GridSize fromInt(final int span)
  {
    for (final GridSize gs : values()) {
      if (gs.getLength() == span) {
        return gs;
      }
    }
    throw new IllegalArgumentException("Impossible de trouver GridSize pour la taille : " + span);
  }
}
