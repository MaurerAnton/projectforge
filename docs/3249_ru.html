package org.projectforge.web.wicket.bootstrap;

/**
 * Используется для определения значения атрибута класса для элементов (размеры сетки Bootstrap).
 * <ul>
 * <li>COL25 — псевдоним для SPAN3</li>
 * <li>COL33 — псевдоним для SPAN4</li>
 * <li>COL50 — псевдоним для SPAN6</li>
 * <li>COL66 — псевдоним для SPAN8</li>
 * <li>COL75 — псевдоним для SPAN9</li>
 * <li>COL100 — псевдоним для SPAN12</li>
 * </ul>
 *
 * @author Кай Райнхард (k.reinhard@micromata.de)
 */
public enum GridSize
{
  SPAN1(1), SPAN2(2), SPAN3(3), SPAN4(4), SPAN6(6), SPAN8(8), SPAN9(9), SPAN12(12), COL25(SPAN3), COL33(SPAN4), COL50(
      SPAN6), COL66(SPAN8), COL75(
          SPAN9), COL100(SPAN12);

  private final String classAttrValue;

  private final int length;

  public String getClassAttrValue()
  {
    return classAttrValue;
  }

  /**
   * @return длина
   */
  public int getLength()
  {
    return length;
  }

  private GridSize(final int length)
  {
    this.length = length;
    // Для сохранения адаптивной ширины на десктопах и планшетах, но вертикального расположения на смартфонах:
    // Замените .col-md-* на .col-sm-*
    //
    // Для сохранения адаптивной ширины на всех устройствах (без вертикального расположения):
    // Замените .col-md-* на .col-xs-*
    this.classAttrValue = "col-sm-" + length;
  }

  private GridSize(final GridSize master)
  {
    this.length = master.length;
    this.classAttrValue = master.classAttrValue;
  }

  public static GridSize fromInt(final int span)
  {
    for (final GridSize gs : values()) {
      if (gs.getLength() == span) {
        return gs;
      }
    }
    throw new IllegalArgumentException("Не удалось найти GridSize для размера: " + span);
  }
}
