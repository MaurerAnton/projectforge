package org.projectforge.development

import org.junit.jupiter.api.Assertions
import org.junit.jupiter.api.Test
import org.slf4j.LoggerFactory
import java.io.File
import java.io.IOException
import java.nio.charset.StandardCharsets

/**
 * Diferentes comprobaciones de archivos fuente.
 * @author Kai Reinhard
 */
class SourceFileCheckerTest {
    /**
     * Comprueba todas las fuentes Kotlin y Java para la declaración correcta de LoggerFactory.getLogger(SourceFileCheckerTest::class.java). Si alguna
     * declaración de Logger difiere del nombre del archivo fuente, la prueba falla.
     * Ejemplo: Foo.java: private final Logger log = LoggerFactory.getLogger(Bar.class); // Bar.class debería ser Foo.class.
     */
    @Test
    @Throws(IOException::class)
    fun checkLoggerDeclarations() {
        var baseDir = File(System.getProperty("user.dir"))
        if (!baseDir.exists() || !baseDir.isDirectory) return
        if (baseDir.name == "projectforge-common") {
            // Verificar el directorio padre para comprobar también los demás módulos de ProjectForge.
            baseDir = baseDir.parentFile
        }
        log.info("Comprobando directorio '${baseDir.absolutePath}...")
        var processedKotlinFiles = 0
        var processedJavaFiles = 0
        val errorFiles = mutableListOf<String>()
        baseDir.walkTopDown().filter { supportedExtensions.contains(it.extension) }.forEach { file ->
            if (file.extension == "kt") {
                ++processedKotlinFiles
            } else {
                ++processedJavaFiles
            }
            val content = file.readText(StandardCharsets.UTF_8)
            val matchResult = loggerRegex.find(content)
            val loggerClass = matchResult?.groups?.get(1)?.value?.trim()
            if (loggerClass != null && file.nameWithoutExtension != loggerClass) {
                errorFiles.add(file.path)
                log.warn("$file declara una clase de logger externa: '$loggerClass'")
            }
        }
        log.info("Se comprobaron $processedKotlinFiles archivos Kotlin y $processedJavaFiles archivos Java para la declaración de Logger.")
        Assertions.assertTrue( errorFiles.isEmpty(), "Archivos fuente con clases de logger no coincidentes encontrados: ${errorFiles.joinToString { it }}")
    }

    companion object {
        private val loggerRegex = """LoggerFactory\s*.\s*getLogger\s*\(\s*(.*)\s*(\.class|:{2}class)""".toRegex()
        private val supportedExtensions = arrayOf("kt", "java")

        // Kotlin:
        private val log = LoggerFactory.getLogger(SourceFileCheckerTest::class.java)
        // Java: private static final Logger log = LoggerFactory.getLogger(SourceFileCheckerTest.class);
    }
}
