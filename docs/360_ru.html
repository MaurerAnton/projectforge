    /*
     * Расчёт, специфичный для Hibernate. Использует значения, переданные в Hibernate Interceptor.onFlushDirty(), для выполнения
     * расчёта.
     *
     * @param propertyNames Массив строк со всеми именами свойств, переданными в метод onFlushDirty. @param
     * previousState Массив объектов, представляющий предыдущее состояние свойств, указанных в массиве propertyNames. @param
     * currentState Массив объектов, представляющий текущее состояние свойств, указанных в массиве propertyNames. @return DeltaSet, представляющий изменения, обнаруженные в состояниях свойств.
     */
    public static DeltaSet calculateDeltaSet(SessionFactory sf, Set<String> validPropertyNames,
                                             Set<String> invalidPropertyNames,
                                             Serializable entityId, Object entity, String[] propertyNames, Object[] previousState, Object[] currentState) {
        if (previousState == null) {
            previousState = new Object[currentState.length];
        }
        if (propertyNames == null || currentState == null) {
            throw new IllegalArgumentException("Все три массива, переданные для вычисления набора изменений, должны быть не равны null");
        }
        if (propertyNames.length != previousState.length && previousState.length != currentState.length) {
            throw new IllegalArgumentException("Все три массива, переданные для вычисления набора изменений, должны быть одинаковой длины");
        }

        DeltaSet deltaSet = new DeltaSet();
        deltaSet.setEntity(entity.getClass());
        deltaSet.setId(entityId);
        try {
            checkProperties(entity, sf, validPropertyNames, invalidPropertyNames, propertyNames, previousState, currentState,
                    deltaSet);
        } catch (Throwable t) {
            log.error("Ошибка при определении набора изменений", t);
        } finally {
            log.debug("Определение набора изменений завершено");
        }
        return deltaSet;
    }
