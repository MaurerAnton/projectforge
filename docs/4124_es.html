// Nombre:            Thumbnav
// Descripción:       Componente para crear navegaciones en miniatura
//
// Componente:        `uk-thumbnav`
//
// Modificador:       `uk-thumbnav-vertical`
//
// Estados:           `uk-active`
//
// ========================================================================


// Variables
// ========================================================================

$thumbnav-margin-horizontal:                     15px !default;
$thumbnav-margin-vertical:                       $thumbnav-margin-horizontal !default;


/* ========================================================================
   Componente: Thumbnav
 ========================================================================== */

/*
 * 1. Permitir que los elementos se envuelvan en la siguiente línea
 * 2. Restablecer lista
 * 3. Canal
 */

.uk-thumbnav {
    display: flex;
    /* 1 */
    flex-wrap: wrap;
    /* 2 */
    margin: 0;
    padding: 0;
    list-style: none;
    /* 3 */
    margin-left: (-$thumbnav-margin-horizontal);
    @if(mixin-exists(hook-thumbnav)) {@include hook-thumbnav();}
}

/*
 * 1. El espacio se asigna únicamente según las dimensiones del contenido: 0 0 auto
 * 2. Canal
 */

.uk-thumbnav > * {
    /* 1 */
    flex: none;
    /* 2 */
    padding-left: $thumbnav-margin-horizontal;
}


/* Elementos
 ========================================================================== */

/*
 * Elementos
 */

.uk-thumbnav > * > * {
    display: inline-block;
    @if(mixin-exists(hook-thumbnav-item)) {@include hook-thumbnav-item();}
}

/* Hover + Focus */
.uk-thumbnav > * > :hover,
.uk-thumbnav > * > :focus {
    outline: none;
    @if(mixin-exists(hook-thumbnav-item-hover)) {@include hook-thumbnav-item-hover();}
}

/* Activo */
.uk-thumbnav > .uk-active > * {
    @if(mixin-exists(hook-thumbnav-item-active)) {@include hook-thumbnav-item-active();}
}


/* Modificador: 'uk-thumbnav-vertical'
 ========================================================================== */

/*
 * 1. Cambiar dirección
 * 2. Canal
 */

.uk-thumbnav-vertical {
    /* 1 */
    flex-direction: column;
    /* 2 */
    margin-left: 0;
    margin-top: (-$thumbnav-margin-vertical);
}

/* 2 */
.uk-thumbnav-vertical > * {
    padding-left: 0;
    padding-top: $thumbnav-margin-vertical;
}


// Hooks
// ========================================================================

@if(mixin-exists(hook-thumbnav-misc)) {@include hook-thumbnav-misc();}
