// Nom :            Thumbnav
// Description :     Composant pour créer des navigations par vignettes
//
// Composant :       `uk-thumbnav`
//
// Modificateur :    `uk-thumbnav-vertical`
//
// États :           `uk-active`
//
// ========================================================================


// Variables
// ========================================================================

$thumbnav-margin-horizontal:                     15px !default;
$thumbnav-margin-vertical:                       $thumbnav-margin-horizontal !default;


/* ========================================================================
   Composant : Thumbnav
 ========================================================================== */

/*
 * 1. Permettre aux éléments de passer à la ligne suivante
 * 2. Réinitialiser la liste
 * 3. Gouttière
 */

.uk-thumbnav {
    display: flex;
    /* 1 */
    flex-wrap: wrap;
    /* 2 */
    margin: 0;
    padding: 0;
    list-style: none;
    /* 3 */
    margin-left: (-$thumbnav-margin-horizontal);
    @if(mixin-exists(hook-thumbnav)) {@include hook-thumbnav();}
}

/*
 * 1. L'espace est alloué uniquement en fonction des dimensions du contenu : 0 0 auto
 * 2. Gouttière
 */

.uk-thumbnav > * {
    /* 1 */
    flex: none;
    /* 2 */
    padding-left: $thumbnav-margin-horizontal;
}


/* Éléments
 ========================================================================== */

/*
 * Éléments
 */

.uk-thumbnav > * > * {
    display: inline-block;
    @if(mixin-exists(hook-thumbnav-item)) {@include hook-thumbnav-item();}
}

/* Survol + Focus */
.uk-thumbnav > * > :hover,
.uk-thumbnav > * > :focus {
    outline: none;
    @if(mixin-exists(hook-thumbnav-item-hover)) {@include hook-thumbnav-item-hover();}
}

/* Actif */
.uk-thumbnav > .uk-active > * {
    @if(mixin-exists(hook-thumbnav-item-active)) {@include hook-thumbnav-item-active();}
}


/* Modificateur : 'uk-thumbnav-vertical'
 ========================================================================== */

/*
 * 1. Changer la direction
 * 2. Gouttière
 */

.uk-thumbnav-vertical {
    /* 1 */
    flex-direction: column;
    /* 2 */
    margin-left: 0;
    margin-top: (-$thumbnav-margin-vertical);
}

/* 2 */
.uk-thumbnav-vertical > * {
    padding-left: 0;
    padding-top: $thumbnav-margin-vertical;
}


// Crochets
// ========================================================================

@if(mixin-exists(hook-thumbnav-misc)) {@include hook-thumbnav-misc();}
