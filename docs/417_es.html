package org.projectforge.business.fibu;

import org.projectforge.common.i18n.I18nEnum;

public enum KundeStatus implements I18nEnum
{

  /** El cliente se encuentra aún en la fase de adquisición. */
  ACQUISISTION("acquisition"), /** Se realizan transacciones activas con el cliente. */
  ACTIVE("active"), /** Ya no existe (temporalmente) relación comercial con el cliente. */
  NONACTIVE("nonactive"), /** La colaboración con el cliente ha finalizado. Es improbable una nueva colaboración. */
  ENDED("ended"), /** El cliente está insolvente, ha cambiado de razón social, etc. */
  NONEXISTENT("nonexistent");

  public static final KundeStatus[] LIST = new KundeStatus[] { ACQUISISTION, ACTIVE, NONACTIVE, ENDED, NONEXISTENT};

  private String key;

  /**
   * La clave se utilizará, por ejemplo, para i18n.
   * @return
   */
  public String getKey()
  {
    return key;
  }

  public String getI18nKey()
  {
    return "fibu.kunde.status." + key;
  };

  KundeStatus(String key)
  {
    this.key = key;
  }

  public boolean isIn(KundeStatus... status)
  {
    for (KundeStatus st : status) {
      if (this == st) {
        return true;
      }
    }
    return false;
  }
}
