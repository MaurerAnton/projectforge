package org.projectforge.business.fibu.kost.reporting;

import org.apache.commons.lang3.StringUtils;

public enum ReportOutputType
{
  PDF("pdf");

  private String key;

  /**
   * @see #ordinal()
   */
  public int getOrdinal()
  {
    return ordinal();
  }

  /**
   * @param s Ключ или строковое представление типа, например "pdf" или "PDF".
   * @return Соответствующий ReportOutputType.
   * @throws UnsupportedOperationException если тип не найден.
   */
  public static ReportOutputType getType(String s)
  {
    if (StringUtils.isEmpty(s)) {
      return null;
    }
    for (ReportOutputType type : ReportOutputType.values()) {
      if (s.equals(type.toString()) || s.equals(type.getKey())) {
        return type;
      }
    }
    throw new UnsupportedOperationException("Неизвестный тип отчёта: '" + s + "'");
  }

  /**
   * Ключ будет использоваться, например, для i18n.
   *
   * @return
   */
  public String getKey()
  {
    return key;
  }

  ReportOutputType(String key)
  {
    this.key = key;
  }

}
