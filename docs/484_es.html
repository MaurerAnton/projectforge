package org.projectforge.business.jobs;

import org.projectforge.business.teamcal.externalsubscription.TeamEventExternalSubscriptionCache;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.scheduling.annotation.Scheduled;
import org.springframework.stereotype.Component;

@Component
public class ExternalCalendarSubscriptionJob {
    private static final org.slf4j.Logger log = org.slf4j.LoggerFactory
            .getLogger(ExternalCalendarSubscriptionJob.class);

    @Autowired
    private TeamEventExternalSubscriptionCache teamEventExternalSubscriptionCache;

    //@Scheduled(cron = "0 */15 * * * *")
    @Scheduled(cron = "${projectforge.cron.externalCalendar}")
    public void execute() {
        new Thread(() -> {
            log.info("Trabajo de suscripciones a calendarios externos iniciado.");
            try {
                teamEventExternalSubscriptionCache.updateCache();
            } catch (final Throwable ex) {
                log.error("Excepción al ejecutar ExternalCalendarSubscriptionJob: " + ex.getMessage());
            }
            log.info("Trabajo de suscripciones a calendarios externos finalizado.");
        }).start();
    }

}
