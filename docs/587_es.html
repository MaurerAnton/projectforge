package org.projectforge.business.teamcal.filter;

import com.thoughtworks.xstream.annotations.XStreamAsAttribute;
import org.joda.time.DateMidnight;

/**
 * Será eliminado cuando el filtro Wicket sea reemplazado por el filtro React en la versión 7 de ProjectForge.
 */
@Deprecated
public abstract class AbstractCalendarFilter implements ICalendarFilter
{
  private static final long serialVersionUID = -2054541010143924205L;

  @XStreamAsAttribute
  private DateMidnight startDate; // No migrar a PFDateTime. Debe ser DateMidnight para la deserialización de filtros obsoletos. Toda la clase será eliminada después de eliminar Wicket.

  @XStreamAsAttribute
  private Integer firstHour = 8;

  @XStreamAsAttribute
  private Boolean slot30;

  @XStreamAsAttribute
  private ViewType viewType;

  public AbstractCalendarFilter()
  {
    startDate = new DateMidnight();
  }

  /**
   * @return el startDate
   */
  public DateMidnight getStartDate()
  {
    return startDate;
  }

  /**
   * @param startDate el startDate a establecer
   * @return this para encadenamiento.
   */
  public AbstractCalendarFilter setStartDate(final DateMidnight startDate)
  {
    if (startDate != null) {
      this.startDate = startDate;
    } else {
      this.startDate = new DateMidnight();
    }
    return this;
  }

   * Si es verdadero, el intervalo es de 30 minutos; de lo contrario, de 15 minutos.
   *
   * @return el slot30
   */
  public boolean isSlot30()

    return slot30 == Boolean.TRUE;
  }

   * @param slot30 el slot30 a establecer
   * @return this para encadenamiento.
   */
  public AbstractCalendarFilter setSlot30(final boolean slot30)
  {
    this.slot30 = slot30;
    return this;
  }
