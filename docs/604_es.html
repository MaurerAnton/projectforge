package org.projectforge.business.user;

import org.projectforge.framework.persistence.user.entities.GroupDO;

/**
 * Grupos de usuario especiales de ProjectForge, como Administradores y Finanzas. Algunas funcionalidades del sistema solo están disponibles para
 * usuarios que son miembros del grupo requerido.
 *
 * @author Kai Reinhard (k.reinhard@micromata.de)
 *
 */
public enum ProjectForgeGroup
{
  /**
   * Administradores del sistema IT de ProjectForge. No ven todas las funcionalidades (como finanzas, libro de pedidos o la
   * vista del gestor de proyectos).
   */
  ADMIN_GROUP("PF_Admin"), //
  CONTROLLING_GROUP("PF_Controlling"), //
  FINANCE_GROUP("PF_Finance"), // Usuarios con acceso completo a las finanzas de la empresa.
  HR_GROUP("PF_HR"), //Usuarios con acceso completo a los empleados y salarios de la empresa.
  MARKETING_GROUP("PF_Marketing"), // Los usuarios de marketing pueden descargar todas las direcciones en formato excel.
  ORGA_TEAM("PF_Organization"), // Los usuarios del equipo de organización tienen acceso a Posteingang y Postausgang.
  PROJECT_ASSISTANT("PF_ProjectAssistant"), // Usuarios con acceso al libro de pedidos (para pedidos asignados).
  PROJECT_MANAGER("PF_ProjectManager"); //Usuarios con acceso a todas las hojas de tiempo (sin detalles) y al libro de pedidos (para pedidos asignados).

  private String key;

  public String getKey()
  {
    return key;
  }

  /**
   * Igual que {@link #getKey()}. Este es el nombre del grupo.
   */
  public String getName()
  {
    return key;
  }

  public boolean matches(final String groupName)
  {
    return key.equals(groupName);
  }

  /**
   * La clave se utilizará, por ejemplo, para i18n.
   *
   * @return
   */
  @Override
  public String toString()
  {
    return key;
  }

  ProjectForgeGroup(final String key)
  {
    this.key = key;
  }

  public static boolean isSystemGroup(GroupDO group) {
    for (ProjectForgeGroup g : ProjectForgeGroup.values()) {
      if (g.key.equals(group.getName())) {
        return true;
      }
    }
    return false;
  }
}
