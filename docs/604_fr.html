package org.projectforge.business.user;

import org.projectforge.framework.persistence.user.entities.GroupDO;

/**
 * Groupes d'utilisateurs spéciaux de ProjectForge, tels que les administrateurs et les finances.
 * Certaines fonctionnalités système sont disponibles uniquement pour les utilisateurs
 * membres du groupe requis.
 *
 * @author Kai Reinhard (k.reinhard@micromata.de)
 *
 */
public enum ProjectForgeGroup
{
  /**
   * Administrateurs système IT de ProjectForge. Ils ne voient pas toutes les fonctionnalités
   * (telles que les finances, le carnet de commandes ou la vue du chef de projet).
   */
  ADMIN_GROUP("PF_Admin"), //
  CONTROLLING_GROUP("PF_Controlling"), //
  FINANCE_GROUP("PF_Finance"), // Utilisateurs ayant un accès complet aux finances de l'entreprise.
  HR_GROUP("PF_HR"), // Utilisateurs ayant un accès complet aux employés et aux salaires de l'entreprise.
  MARKETING_GROUP("PF_Marketing"), // Les utilisateurs marketing peuvent télécharger toutes les adresses au format Excel.
  ORGA_TEAM("PF_Organization"), // Les utilisateurs de l'équipe organisation ont accès au courrier entrant et sortant.
  PROJECT_ASSISTANT("PF_ProjectAssistant"), // Utilisateurs ayant accès au carnet de commandes (pour les commandes assignées).
  PROJECT_MANAGER("PF_ProjectManager"); // Utilisateurs ayant accès à toutes les feuilles de temps (sans détails) et au carnet de commandes (pour les commandes assignées).

  private String key;

  public String getKey()
  {
    return key;
  }

  /**
   * Identique à {@link #getKey()}. C'est le nom du groupe.
   */
  public String getName()
  {
    return key;
  }

  public boolean matches(final String groupName)
  {
    return key.equals(groupName);
  }

  /**
   * La clé sera utilisée par exemple pour l'i18n.
   *
   * @return
   */
  @Override
  public String toString()
  {
    return key;
  }

  ProjectForgeGroup(final String key)
  {
    this.key = key;
  }

  public static boolean isSystemGroup(GroupDO group) {
    for (ProjectForgeGroup g : ProjectForgeGroup.values()) {
      if (g.key.equals(group.getName())) {
        return true;
      }
    }
    return false;
  }
}
