package org.projectforge.business.user;

import org.projectforge.common.i18n.I18nEnum;

/**
 * Die Kategorie wird nur zur Sortierung der Benutzerrechte für die Bearbeitung in der GUI verwendet.
 * 
 * @author Kai Reinhard (k.reinhard@micromata.de)
 * 
 */
public enum UserRightCategory implements I18nEnum
{
  ADMIN("admin"), FIBU("fibu"), MISC("misc"), ORGA("orga"), HR("hr"),
  /** Projektmanagement */
  PM("pm"), PLUGINS("plugins");

  private String key;

  /**
   * Der Schlüssel wird z. B. für die Internationalisierung verwendet.
   * 
   * @return
   */
  public String getKey()
  {
    return key;
  }

  public String getI18nKey()
  {
    return "access.right.category" + key;
  }

  UserRightCategory(final String key)
  {
    this.key = key;
  }
}
