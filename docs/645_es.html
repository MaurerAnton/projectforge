package org.projectforge.export;

/**
 * Representa un objeto de exportación incluyendo el nombre de archivo deseado.
 * 
 * @author Kai Reinhard (k.reinhard@micromata.de)
 * 
 */
public class ExportZipFile
{
  private final String filename;

  private final Object exportObject;

  public ExportZipFile(final String filename, final Object exportObject)
  {
    this.filename = filename;
    this.exportObject = exportObject;
  }

  /**
   * @return el nombre de archivo
   */
  public String getFilename()
  {
    return filename;
  }

  /**
   * @return el objeto de exportación
   */
  public Object getExportObject()
  {
    return exportObject;
  }
}
