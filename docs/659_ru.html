package org.projectforge.framework.access;

import org.projectforge.common.i18n.I18nEnum;

/**
 * TODO Ошибка проектирования: Домены должны быть явно перечислены.
 * 
 * @author Kai Reinhard (k.reinhard@micromata.de)
 * 
 */
public enum AccessType implements I18nEnum
{
  /**
   * Требуется некоторыми методами hasAccess, если проверяется только связь с группой. Используется для конструктора
   * UserException.
   */
  GROUP("group"),
  /** Права доступа для select, insert, update, delete задач. */
  TASKS("tasks"),

  TASK_ACCESS_MANAGEMENT("accessManagement"),

  TIMESHEETS("timesheets"),

  OWN_TIMESHEETS("ownTimesheets");

  private String key;

  /**
   * @return Суффикс ключа будет использоваться, например, для i18n.
   */
  public String getKey()
  {
    return key;
  }

  /**
   * @return Полный ключ i18n, включая префикс i18n "fibu.auftrag.status.".
   */
  @Override
  public String getI18nKey()
  {
    return "access.type." + key;
  }

  AccessType(final String key)
  {
    this.key = key;
  }
}
