package org.projectforge.storage;

import org.projectforge.framework.configuration.ConfigXml;

/**
 * Bean, die von ConfigXML (config.xml) zur Konfiguration des Speichers (JackRabbit-Repository) verwendet wird. Der Speicher läuft auf einem separaten Webserver oder als
 * eigenes WAR auf demselben Webserver.
 * @author Kai Reinhard (k.reinhard@micromata.de)
 */
public class StorageConfig
{
  private String url;

  private String authenticationToken;

  /**
   * @return die URL der Speicher-Web-App.
   */
  public String getUrl()
  {
    return url;
  }

  /**
   * @param url die zu setzende URL
   * @return this für Verkettung.
   */
  public StorageConfig setUrl(final String url)
  {
    this.url = url;
    return this;
  }

  /**
   * @return das Authentifizierungstoken, das für die Kommunikation zwischen der ProjectForge-Web-App und der Speicher-Web-App verwendet werden soll.
   */
  public String getAuthenticationToken()
  {
    return authenticationToken;
  }

  /**
   * @param authenticationToken das zu setzende Authentifizierungstoken
   * @return this für Verkettung.
   */
  public StorageConfig setAuthenticationToken(final String authenticationToken)
  {
    this.authenticationToken = authenticationToken;
    return this;
  }

  /**
   * @see ConfigXml#toString(Object)
   */
  @Override
  public String toString()
  {
    return ConfigXml.toString(this);
  }
}
