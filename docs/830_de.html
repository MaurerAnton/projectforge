    // Telefon-Regex (verschiedene Formate mit flexiblen Trennzeichen)
    // Erkennt Telefonnummern mit Ziffern und üblichen Trennzeichen (Leerzeichen, -, /, ., Klammern)
    // Unterstützt sowohl "Tel:" als auch "Tel" (mit/ohne Doppelpunkt)
    private val PHONE_REGEX = Regex(
        """(?:Tel\.?:?|Telefon:?|Phone:?|Fon:?|Mobil:?|Mobile:?|Fax:?)\s*(\+?(?:\d+[\s\-./()]*)+\d)""",
        RegexOption.IGNORE_CASE
    )

    // Nackte Telefonnummern (Telefonnummer ohne Präfix-Label)
    // Erkennt Telefonnummern, die mit + oder Ländervorwahl beginnen und mindestens 8 Ziffern haben
    // Dies vermeidet Fehlalarme bei regulären Zahlen
    private val BARE_PHONE_REGEX = Regex(
        """^\+?(?:\d+[\s\-./()]*){8,}$"""
    )

    // PLZ + Ort (4-5 Ziffern + Ortsname, optional mit "D-" oder "CH-" Präfix)

        """(?:D-|CH-)?(\d{4,5})\s+([A-ZÄÖÜ][a-zäöüß]+(?:[\s-][A-ZÄÖÜ]?[a-zäöüß]+)*)""",
    )

    // Straßenadresse (Straßenname + Hausnummer)
    private val STREET_REGEX = Regex(
        """([A-ZÄÖÜ][a-zäöüß]+(?:[\s-][A-ZÄÖÜ]?[a-zäöüß]+)*\.?(?:\s+|-)(?:\d+[a-zA-Z]?(?:\s*-\s*\d+[a-zA-Z]?)?))""",
    )

    // Ländername (häufige Länder in mehreren Sprachen, optional mit zweitem Namen nach /)

        RegexOption.IGNORE_CASE
